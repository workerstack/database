// sqlite/constraints.d.ts
/**
 * Type-safe column constraint definitions
 * These exist only in the type system and are completely erased at runtime
 */
import type * as z from "zod";
import type { IntegerFormat, TextFormat } from "../formats/catalog.ts";
/** What every helper's constraints take: the DTO, mutation, privacy, relation and tenant flags */
type BaseConstraints = {
    /** Field is filled by comb or the database: generated create and update DTOs leave it out */
    autogenerate?: boolean | undefined;
    /**
     * A foreign key that may move after creation (a person to another team): generated update DTOs
     * take it. Every other reference is set once, at creation. Refused on a column that references
     * nothing, and together with `nomutate`.
     */
    mutable?: boolean | undefined;
    /** Field cannot be mutated after creation */
    nomutate?: boolean | undefined;
    /**
     * Field is never read back: generated read DTOs and the meta stamp leave it out, and the manifest
     * lists it flagged `private: true`, on both dialects
     */
    private?: boolean | undefined;
    /**
     * the name this foreign key's relation and its way back take, where the column's role would take a
     * column's own name (`currency` toward `currency.code`, a `parent` self-reference): an identifier,
     * refused on a column that references nothing
     */
    relation?: string | undefined;
    /**
     * This column scopes rows to a tenant.
     *
     * Declared, never inferred: comb can see that a column is a foreign key but
     * not that the table it points at is the tenant, and a wrong tenant boundary
     * is worse than a missing one. Saying so here makes it a hand-reviewed fact.
     * At most one column per table may carry it.
     */
    tenant?: boolean | undefined;
};
/**
 * Rewrites a written text takes in the generated DTOs before its checks, so its bounds and pattern
 * hold the rewritten value: `trim` first, then `lowercase`, or `uppercase` when `lowercase` is unset.
 * An email is always trimmed and lowercased, a url trimmed and stripped of tabs and line breaks, and
 * codegen refuses a rewrite either would drop, or any on a password, which is kept as written
 */
type StringTransformConstraints = {
    /** Apply lowercase transformation */
    lowercase?: boolean | undefined;
    /** Trim whitespace */
    trim?: boolean | undefined;
    /** Apply uppercase transformation */
    uppercase?: boolean | undefined;
};
/** Bounds a written value is held to: a text's length, or a number's or timestamp's value */
type NumericRangeConstraints = {
    /** Maximum value */
    max?: number | undefined;
    /** Minimum value */
    min?: number | undefined;
};
/**
 * What a written text must hold, checked by the generated DTOs: one kind at most, of `email`, `url`,
 * `password`, `imageUri` and `name`, which codegen refuses a second of, and a `pattern` beside it
 */
type StringValidationConstraints = {
    /** Email validation */
    email?: boolean | undefined;
    /** Image URI validation (accepts https URLs and data:image/ URIs) */
    imageUri?: boolean | undefined;
    /** Human/entity name validation (letters, marks, numbers, spaces, punctuation) */
    name?: boolean | undefined;
    /** A stored password hash: kept as written, held to the declared bounds and pattern alone; validate the plaintext with `passwordText()` */
    password?: boolean | undefined;
    /** Regex pattern validation */
    pattern?: string | undefined;
    /** URL validation */
    url?: boolean | undefined;
};
/**
 * Text column constraints
 * Closed record - only these properties allowed
 */
export type TextConstraints = BaseConstraints & StringTransformConstraints & NumericRangeConstraints & StringValidationConstraints & {
    /**
     * The standard a written text holds: the generated DTOs run the format's trim and case, then hold the
     * value to its pattern and, where it names one, its pinned registry, and publish the id as `x-comb-format`.
     * A value it refuses is one zod `invalid_format` issue carrying the id, worded by zod's locale. One of:
     * - `base64url`: RFC 4648 §5 base64url, unpadded; `dGVzdA`; pattern only
     * - `bcp47`: BCP 47 language tag, RFC 5646 §2.1 well-formed; `zh-Hant-TW`; pattern only
     * - `e164`: ITU-T E.164 phone number, + and up to 15 digits; `+385911234567`; pattern only
     * - `iana-time-zone`: IANA time zone name, tzdata 2026f; `Europe/Zagreb`; registry
     * - `ip`: IPv4 dotted quad or IPv6 address, RFC 4291 §2.2; `2001:db8::1`; pattern only
     * - `ipv4`: IPv4 address, dotted quad without leading zeros; `192.0.2.1`; pattern only
     * - `ipv6`: IPv6 address, RFC 4291 §2.2; `2001:db8::1`; pattern only
     * - `iso3166-1-alpha-2`: ISO 3166-1 alpha-2 country code, Debian iso-codes v4.20.1; `HR`; registry
     * - `iso3166-1-alpha-3`: ISO 3166-1 alpha-3 country code, Debian iso-codes v4.20.1; `HRV`; registry
     * - `iso4217`: ISO 4217 alphabetic currency code, SIX list 2026-09-17; `EUR`; registry
     * - `iso639-1`: ISO 639-1 language code, Debian iso-codes v4.20.1; `hr`; registry
     * - `kebab-slug`: kebab-case slug, lowercase letters and digits joined by single hyphens; `spring-sale-2026`; pattern only
     * - `pkce-s256-challenge`: RFC 7636 §4.2 PKCE SHA-256 code challenge; `E9Melhoa2OwvFrEMTJguCHaoeK1t8URWbuGJSstw-cM`; pattern only
     * - `rfc1123-hostname`: RFC 1123 §2.1 host name; `api.example.com`; pattern only
     * - `semver`: Semantic Versioning 2.0.0 version; `1.4.0-rc.1`; pattern only
     * - `sha256-hex`: SHA-256 digest, FIPS 180-4, lowercase hex; `e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855`; pattern only
     * - `sha512-hex`: SHA-512 digest, FIPS 180-4, lowercase hex; `cf83e1357eefb8bdf1542850d66d8007d620e4050b5715dc83f4a921d36ce9ce47d0d13c5d85f2b0ff8318d2877eec2f63b931bd47417a81a538327af927da3e`; pattern only
     * - `srgb-hex`: sRGB color as #rrggbb, CSS Color 4; `#1e90ff`; pattern only
     * - `srgb-hex-alpha`: sRGB color with alpha as #rrggbbaa, CSS Color 4; `#1e90ff80`; pattern only
     * - `ulid`: ULID, Crockford base32 in lowercase; `01hzy3k8q2m4n5p6r7s8t9v0wx`; pattern only
     * - `uts35-ca`: UTS #35 calendar (ca) key value, CLDR 49; `gregory`; pattern only
     * - `uts35-fw`: UTS #35 first day of the week (fw) key value, CLDR 49; `mon`; pattern only
     * - `uts35-hc`: UTS #35 hour cycle (hc) key value, CLDR 49; `h23`; pattern only
     * - `uts35-ms`: UTS #35 measurement system (ms) key value, CLDR 49; `metric`; pattern only
     * - `uts35-mu`: UTS #35 measurement unit (mu) key value, CLDR 49; `celsius`; pattern only
     * - `uts35-nu`: UTS #35 numbering system (nu) key value, CLDR 49; `latn`; pattern only
     * - `uuid`: RFC 9562 UUID, lowercase hex; `0f8fad5b-d9cb-469f-a165-70867728950e`; pattern only
     * - `uuid-v4`: RFC 9562 UUID version 4, lowercase hex; `0f8fad5b-d9cb-469f-a165-70867728950e`; pattern only
     * - `uuid-v7`: RFC 9562 UUID version 7, lowercase hex; `01890a5d-ac96-774b-bcce-b302099a8057`; pattern only
     */
    format?: TextFormat | undefined;
    /**
     * An exported zod schema the generated DTOs pipe the value into after every other rule, on the
     * transformed value; its issues reach the caller as zod yields them, and its input JSON Schema is
     * published beside `x-comb-schema`. Codegen imports it as it imports a `c.json` schema, so export it
     * from a module of its own, or above the table that names it
     */
    schema?: z.ZodType<string, string> | undefined;
};
/**
 * Floating-point column constraints: its bounds, and an exported zod schema its value is piped into
 * Closed record - only these properties allowed
 */
export type RealConstraints = BaseConstraints & NumericRangeConstraints & {
    /**
     * An exported zod schema the generated DTOs pipe the value into after its bounds; its issues reach
     * the caller as zod yields them, and its input JSON Schema is published beside `x-comb-schema`.
     * Codegen imports it as it imports a `c.json` schema, so export it from a module of its own, or
     * above the table that names it
     */
    schema?: z.ZodType<number, number> | undefined;
};
/**
 * Integer column constraints: a real's, and the standard `format` that numbers its values
 * Closed record - only these properties allowed
 */
export type IntegerConstraints = RealConstraints & {
    /**
     * The standard that numbers the value: its range bounds the generated DTOs, the tighter of it and a
     * declared `min` or `max` winning, and they publish the id as `x-comb-format`. One of:
     * - `iso8601-month`: ISO 8601 month number, 1 January to 12 December; `3`
     * - `iso8601-weekday`: ISO 8601 §3.2.2 weekday number, 1 Monday to 7 Sunday; `1`
     */
    format?: IntegerFormat | undefined;
};
/**
 * Foreign key column constraints: a text's but `format` and `schema`, as the value is another
 * table's key, held to that key's own rules
 * Closed record - only these properties allowed
 */
export type RefConstraints = Omit<TextConstraints, "format" | "schema">;
/**
 * JSON column constraints
 * Closed record - only these properties allowed
 */
export type JsonConstraints = BaseConstraints & {
    /** Maximum JSON stringified byte length */
    maxBytes?: number | undefined;
};
/**
 * Timestamp column constraints
 * Closed record - only these properties allowed
 */
export type TimestampConstraints = BaseConstraints & NumericRangeConstraints;
/**
 * State machine declared alongside an enum column's values.
 *
 * Type-level only, like every other constraint here — the analyzer reads it
 * out of the source. Hold the same object as a const and pass it to both
 * `c.enum` and `assertTransition`. See docs/state-machines.md.
 */
export type EnumStates<TState extends string = string> = {
    /** The value a row starts at */
    initial?: TState | undefined;
    /** Values from which no transition is legal */
    terminal?: readonly TState[] | undefined;
    /** Legal moves, `from` → `to[]`. Omitted states are unconstrained. */
    transitions?: {
        readonly [K in TState]?: readonly TState[] | undefined;
    } | undefined;
};
/**
 * What `c.enum` takes beside its values: its state machine, the `nomutate`, `private` and `tenant`
 * flags every helper takes, and the standard `format` its values are written in, which codegen holds
 * each value to and publishes as `x-comb-format`
 */
export type EnumOptions<TState extends string = string> = EnumStates<TState> & Pick<BaseConstraints, "nomutate" | "private" | "tenant"> & {
    /**

     * - `iso3166-1-alpha-2`: ISO 3166-1 alpha-2 country code, Debian iso-codes v4.20.1; `HR`; registry
     * - `iso3166-1-alpha-3`: ISO 3166-1 alpha-3 country code, Debian iso-codes v4.20.1; `HRV`; registry
     * - `iso4217`: ISO 4217 alphabetic currency code, SIX list 2026-09-17; `EUR`; registry
     * - `iso639-1`: ISO 639-1 language code, Debian iso-codes v4.20.1; `hr`; registry
     * - `kebab-slug`: kebab-case slug, lowercase letters and digits joined by single hyphens; `spring-sale-2026`; pattern only
     * - `pkce-s256-challenge`: RFC 7636 §4.2 PKCE SHA-256 code challenge; `E9Melhoa2OwvFrEMTJguCHaoeK1t8URWbuGJSstw-cM`; pattern only
     * - `rfc1123-hostname`: RFC 1123 §2.1 host name; `api.example.com`; pattern only
     * - `semver`: Semantic Versioning 2.0.0 version; `1.4.0-rc.1`; pattern only

     * - `sha512-hex`: SHA-512 digest, FIPS 180-4, lowercase hex; `cf83e1357eefb8bdf1542850d66d8007d620e4050b5715dc83f4a921d36ce9ce47d0d13c5d85f2b0ff8318d2877eec2f63b931bd47417a81a538327af927da3e`; pattern only

// formats/catalog.d.ts
/** The ids `c.text` and `c.enum` take as `format`, sorted by code unit */
export declare const TEXT_FORMATS: readonly ["base64url", "bcp47", "e164", "iana-time-zone", "ip", "ipv4", "ipv6", "iso3166-1-alpha-2", "iso3166-1-alpha-3", "iso4217", "iso639-1", "kebab-slug", "pkce-s256-challenge", "rfc1123-hostname", "semver", "sha256-hex", "sha512-hex", "srgb-hex", "srgb-hex-alpha", "ulid", "uts35-ca", "uts35-fw", "uts35-hc", "uts35-ms", "uts35-mu", "uts35-nu", "uuid", "uuid-v4", "uuid-v7"];
/** The ids `c.integer` takes as `format`, sorted by code unit */
export declare const INTEGER_FORMATS: readonly ["iso8601-month", "iso8601-weekday"];
/** A text format's id */
export type TextFormat = (typeof TEXT_FORMATS)[number];
/** An integer format's id */
export type IntegerFormat = (typeof INTEGER_FORMATS)[number];
/** The registry predicate a text format's value must also satisfy */
type Member = "isIanaTimeZone" | "isIso3166Alpha2" | "isIso3166Alpha3" | "isIso4217" | "isIso639Alpha2";
/** What a text format checks, after its transforms, on the value its DTO is given */
export type FormatSpec = {
    /** The standard and the release it follows, as a person reads it */
    readonly standard: string;
    /** The anchored pattern the value matches, portable to RE2 and JavaScript's `u` flag */
    readonly source: string;
    /** The fewest characters a value has */
    readonly min: number;
    /** The most characters a value has, or null when the format sets no cap and a column declares one */
    readonly max: number | null;
    /** The transforms that run, in order, before the check */
    readonly transform: readonly ("lowercase" | "trim" | "uppercase")[];
    /** The pinned registry the value must belong to, by its predicate's name, or null when the source is the whole rule */
    readonly member: Member | null;
    /** A value of the format as written, which the format takes and a generated DTO publishes as its example */
    readonly example: string;
};
/** The range an integer format's value lies in */
export type IntegerSpec = {
    /** The standard that numbers the values, as a person reads it */
    readonly standard: string;
    /** The least value */
    readonly min: number;
    /** The greatest value */
    readonly max: number;
};
/** Every text format's spec, by id */
export declare const FORMATS: {
    readonly [F in TextFormat]: FormatSpec;
};
/** Every integer format's range, by id */
export declare const INTEGER_SPECS: {
    readonly [F in IntegerFormat]: IntegerSpec;
};
/**
 * Whether `format` takes `value` exactly as written, its transforms not run: within the format's
 * length bounds, matching its source, and a member of its registry when it names one. Never throws.
 */
export declare function formatTakes(format: TextFormat, value: string): boolean;
export {};
